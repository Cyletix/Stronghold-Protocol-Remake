@echo off
setlocal

set "HOST=127.0.0.1"
set "PORT=8000"
if not "%~1"=="" set "HOST=%~1"
if not "%~2"=="" set "PORT=%~2"

set "PY=python"
if exist ".venv\\Scripts\\python.exe" set "PY=.venv\\Scripts\\python.exe"
where "%PY%" >nul 2>&1 || (
  echo 未找到可用的 python 可执行文件，请先建立/激活 .venv 或安装 Python。
  pause
  exit /b 1
)

cd /d "%~dp0"
start "" "http://%HOST%:%PORT%/"
"%PY%" tools\run_server.py --host %HOST% --port %PORT%
